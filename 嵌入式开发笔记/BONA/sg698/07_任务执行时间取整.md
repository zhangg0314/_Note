void CTaskEvent::InitByTaskStartTime(const oop_task_t& task)
{
   memcpy(&m_task, &task, sizeof(oop_task_t));
   m_last_active_time = 0;
   if (NORAML == task.GatherType)
   {
      normalgather_t gather;
      if (0 != GetNormalGather(task.GatherID, gather))
      {
         LOG(WARN_L, "normal task %d has no corresponding task scheme %d\n", task.ID, task.GatherID);
      }
   }
   else if (EVENT == task.GatherType)
   {
      oop_evtgather_t gather;
      if (0 != GetEventGather(task.GatherID, gather))
      {
         LOG(WARN_L, "event task %d has no corresponding event task scheme %d\n", task.ID, task.GatherID);
      }
   }
   else if (REPORT == task.GatherType)
   {
      oop_report_t gather;
      memset(&gather, 0, sizeof(gather));
      if (0 !=  GetReportGather(task.GatherID, gather))
      {
         LOG(WARN_L, "report task %d has no corresponding event task scheme %d\n", task.ID, task.GatherID);
      }
   }
   else
   {
      LOG(WARN_L, "GatherType is TRANS\n");
   }
   time_t taskstart = 0;
   time_t minstart = 0;
   CGX698DateTime::ToLinuxTm(task.StartTime, taskstart);
   CGX698DateTime::ToLinuxTm(task.EndTime, m_task_expiretime);
   struct tm temp = *localtime(&taskstart);
   time_t now = time(NULL);
   struct tm taskstarttm;
   if (taskstart > now) //在将来
   {
       
      // LOG(WARN_L, "future!\n");
      minstart = taskstart;
      taskstarttm = temp;
   }
   else
   {
     //  LOG(WARN_L, "now!\n");
      minstart = now; //现在
      taskstarttm = *localtime(&now);
   }
   switch (task.Interval.scale)
   {
   case SECOND:
       m_activeTime = minstart - (minstart % task.Interval.value);
       break;
   case MINUTE:
      {       
    #if  defined(LOCALE_SHANXI)   
	   //山西1小时曲线任务07表实时数据间隔15分钟(这些特殊处理下一个版本删除)
	   if (task.GatherType == 1 && task.Interval.value == 60 && access("/home/app/conf/instant2freeze",R_OK) == 0)
	   {
	    	m_activeTime = minstart - (minstart % (15 * SECONDS_MINUTE));
	   }
	   else
	   {
	      m_activeTime = minstart - (minstart % (task.Interval.value * SECONDS_MINUTE));
       }
#else
           m_activeTime = minstart - (minstart % (task.Interval.value * SECONDS_MINUTE));

 #endif

     

      }
      break;
   case HOUR:
      {
         taskstarttm.tm_min = temp.tm_min;
         taskstarttm.tm_sec = 0;
         m_activeTime = mktime(&taskstarttm);
         struct tm activetm = *localtime(&m_activeTime);
         while (0 != (activetm.tm_hour % task.Interval.value))
         {
            m_activeTime -= SECONDS_HOUR;
            activetm = *localtime(&m_activeTime);
         }
#if defined(LOCALE_SHANXI)
		
         if (REPORT == task.GatherType && 12 == task.Interval.value)
         {
            m_activeTime += (temp.tm_hour % 12) * 3600;
         }
		
		else if(1 == task.GatherType && 1 == task.Interval.value && access("/home/app/conf/instant2freeze",R_OK) == 0)
		{
		//山西1小时曲线任务07表实时数据间隔15分钟
	        m_activeTime = minstart - (minstart % (15 * SECONDS_MINUTE));
		}
#endif
        }
        break;
    case DAY:
       taskstarttm.tm_hour = temp.tm_hour;
       taskstarttm.tm_min = temp.tm_min;
#if defined(LOCALE_STANDARD)
        
        if (task.Timenum > 0) 
        {
            if (task.TaskTimeFrame[0].B_HOUR > temp.tm_hour)
            {
                taskstarttm.tm_hour = task.TaskTimeFrame[0].B_HOUR;
            }

            if (task.TaskTimeFrame[0].B_MIN > temp.tm_min)
            {
                taskstarttm.tm_min = task.TaskTimeFrame[0].B_MIN;
            }
        }
#endif
        taskstarttm.tm_sec = 0;
        m_activeTime = mktime(&taskstarttm);
        break;
    case MONTH:
        taskstarttm.tm_mday = temp.tm_mday;
        taskstarttm.tm_hour = temp.tm_hour;
        taskstarttm.tm_min = temp.tm_min;
        taskstarttm.tm_sec = 0;
        m_activeTime = mktime(&taskstarttm);
        break;
    default:
        //assert(0 == true);
        break;
    }
   m_activeTime += TimeUtils::CalcDelaySecs(task.Delay);

	//河北为满足整15分钟前10s优先执行1分钟任务，将普通采集任务生成时间延后或提前,1分钟提前2s
	#if defined(LOCALE_HEBEI)
        if(m_task.GatherType ==1 && m_task.Interval.scale == MINUTE && m_task.Interval.value == 1)

	#endif
