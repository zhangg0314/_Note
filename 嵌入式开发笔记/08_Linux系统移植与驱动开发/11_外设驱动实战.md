# 外设驱动实战

## 1. 开发板启动过程

以 Exynos4412 为例：

1. 开发板上电后，首先第一个运行的程序是处于地址 `0x00000000` 的 SOC 内部 iROM（内部只读存储器）中固化的程序（BL0），它初始化基本的系统功能，如时钟和堆栈。
2. iROM 从一个特定的启动设备加载 BL1 映像到内部 256KB SRAM。即通过 OM（Operating Mode）引脚选择启动设备。根据安全启动键值，iROM 对 BL1 映像进行完整性检查。
3. BL1 初始化系统时钟和 DRAM 控制器。初始化 DRAM 控制器后，将 OS 映像从引导设备加载到 DRAM 中。根据安全启动密钥值，BL1 可以对操作系统映像做一个完整性检查。
4. 引导完成后，BL1 跳转到操作系统。

## 2. 访问外设控制器的寄存器（GPIO）

IO 访问即访问外设控制器的寄存器，寄存器的地址空间也算作内核空间中的一部分。

### 2.1 IO 访问两种方式

1. **IO 端口**：X86 上用 IO 指令访问
2. **IO 内存**：外设寄存器在 SOC 芯片手册上都有相应物理内存地址，不是虚拟内存地址，即外设寄存器也作为 SOC 内存的一部分。

### 2.2 IO 内存访问接口

```c
static inline void __iomem *ioremap(unsigned long offset, unsigned long size);
// 功能：实现 IO 管脚的映射
// 参数：offset - 该管脚的偏移地址，size - 该管脚映射空间的大小
// 返回值：成功返回映射的虚拟地址，失败 NULL

static inline void iounmap(volatile void __iomem *addr);
// 功能：解除 IO 管脚的映射

unsigned readb(void *addr);   // 1字节 或 ioread8
unsigned readw(void *addr);   // 2字节 或 ioread16
unsigned readl(void *addr);   // 4字节 或 ioread32

void writeb(unsigned value, void *addr);  // 1字节 或 iowrite8
void writew(unsigned value, void *addr);  // 2字节 或 iowrite16
void writel(unsigned value, void *addr);  // 4字节 或 iowrite32
```


- 考虑需要支持的函数

- 模块出口：iounmap
- 编写关灯函数和开灯函数，实现 ioctl

## 4. 裸机开发：链接脚本

```c
OUTPUT_FORMAT("elf32-littlearm", "elf32-littlearm", "elf32-littlearm")
OUTPUT_ARCH(arm)
ENTRY(_start)
SECTIONS
{
    . = 0x40008000;
    . = ALIGN(4);
    .text :
    {
        start/start.o(.text)
        *(.text)
    }
    . = ALIGN(4);
    .rodata :
    { *(.rodata) }
    . = ALIGN(4);
    .data :
    { *(.data) }
    . = ALIGN(4);
    .bss :
    { *(.bss) }
}
```

链接脚本说明：
- `OUTPUT_FORMAT`：指定输出文件的 ELF 格式（小端 ARM）
- `OUTPUT_ARCH`：目标架构为 ARM
- `ENTRY(_start)`：程序入口点为 `_start` 符号
- `. = 0x40008000`：代码段起始地址
- `. = ALIGN(4)`：4 字节对齐
- `.text`/`.rodata`/`.data`/`.bss`：依次放置代码段、只读数据段、数据段、BSS 段

## 5. 项目示例：platform LED 驱动（设备树匹配）

**设备树节点**：

```dts
leds {
    compatible = "my,leds";
    led2-gpio = <&gpx2 7 0>;
    led3-gpio = <&gpx1 0 0>;
    led4-gpio = <&gpf3 4 0>;
    led5-gpio = <&gpf3 5 0>;
};
```

**驱动代码**：

```c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/platform_device.h>
#include <linux/of_gpio.h>
#include <linux/gpio.h>
#include <linux/cdev.h>
#include <linux/fs.h>
#include <linux/uaccess.h>

#define LED_NUM 4

struct led_dev {
    int gpios[LED_NUM];
    struct cdev cdev;
    dev_t devno;
};

#define LED_MAGIC 'L'
#define LED_ON  _IOW(LED_MAGIC, 1, int)
#define LED_OFF _IOW(LED_MAGIC, 2, int)

static long led_ioctl(struct file *file, unsigned int cmd, unsigned long arg)
{
    struct led_dev *dev = file->private_data;
    int led_index = (int)arg;

    if (led_index < 0 || led_index >= LED_NUM)
        return -EINVAL;

    switch (cmd) {
    case LED_ON:
        gpio_set_value(dev->gpios[led_index], 0);
        break;
    case LED_OFF:
        gpio_set_value(dev->gpios[led_index], 1);
        break;
    default:
        return -ENOTTY;
    }
    return 0;
}

static int led_open(struct inode *inode, struct file *file)
{
    file->private_data = container_of(inode->i_cdev, struct led_dev, cdev);
    return 0;
}

static const struct file_operations led_fops = {
    .owner = THIS_MODULE,
    .open = led_open,
    .unlocked_ioctl = led_ioctl,
};

static int led_probe(struct platform_device *pdev)
{
    struct device_node *np = pdev->dev.of_node;
    struct led_dev *dev;
    int ret, i;
    char gpio_name[16];

    dev = devm_kzalloc(&pdev->dev, sizeof(*dev), GFP_KERNEL);
    if (!dev) return -ENOMEM;

    for (i = 0; i < LED_NUM; i++) {
        snprintf(gpio_name, sizeof(gpio_name), "led%d-gpio", i + 2);
        dev->gpios[i] = of_get_named_gpio(np, gpio_name, 0);
        if (!gpio_is_valid(dev->gpios[i])) {
            dev_err(&pdev->dev, "invalid gpio: %s\n", gpio_name);
            return -EINVAL;
        }
        ret = gpio_request(dev->gpios[i], gpio_name);
        if (ret) return ret;
        gpio_direction_output(dev->gpios[i], 1);
    }

    ret = alloc_chrdev_region(&dev->devno, 0, 1, "leds");
    if (ret) goto err_gpio;

    cdev_init(&dev->cdev, &led_fops);
    dev->cdev.owner = THIS_MODULE;
    ret = cdev_add(&dev->cdev, dev->devno, 1);
    if (ret) {
        unregister_chrdev_region(dev->devno, 1);
        goto err_gpio;
    }

    platform_set_drvdata(pdev, dev);
    dev_info(&pdev->dev, "LED driver probed, major=%d\n", MAJOR(dev->devno));
    return 0;

err_gpio:
    for (i--; i >= 0; i--)
        gpio_free(dev->gpios[i]);
    return ret;
}

static int led_remove(struct platform_device *pdev)
{
    struct led_dev *dev = platform_get_drvdata(pdev);
    int i;
    cdev_del(&dev->cdev);
    unregister_chrdev_region(dev->devno, 1);
    for (i = 0; i < LED_NUM; i++)
        gpio_free(dev->gpios[i]);
    return 0;
}

static const struct of_device_id led_of_match[] = {
    { .compatible = "my,leds" },
    {}
};
MODULE_DEVICE_TABLE(of, led_of_match);

static struct platform_driver led_driver = {
    .probe  = led_probe,
    .remove = led_remove,
    .driver = {
        .name = "led_driver",
        .of_match_table = led_of_match,
    },
};
module_platform_driver(led_driver);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("LED driver with platform bus and device tree");
```

**应用层测试**：

```c
#include <stdio.h>
#include <fcntl.h>
#include <sys/ioctl.h>

#define LED_MAGIC 'L'
#define LED_ON  _IOW(LED_MAGIC, 1, int)
#define LED_OFF _IOW(LED_MAGIC, 2, int)

int main(int argc, char *argv[])
{
    int fd, led_index;
    fd = open("/dev/leds", O_RDWR);
    if (fd < 0) { perror("open"); return -1; }

    for (led_index = 0; led_index < 4; led_index++) {
        ioctl(fd, LED_ON, led_index);
        usleep(500000);
        ioctl(fd, LED_OFF, led_index);
    }
    close(fd);
    return 0;
}
```

## 6. 项目示例：GPIO 按键驱动 + input 子系统

```c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/interrupt.h>
#include <linux/input.h>
#include <linux/gpio.h>
#include <linux/of_gpio.h>
#include <linux/platform_device.h>
#include <linux/timer.h>

struct key_dev {
    int gpio;
    int irq;
    int key_code;
    struct input_dev *input_dev;
    struct timer_list timer;
};

static void key_timer_callback(struct timer_list *t)
{
    struct key_dev *dev = from_timer(dev, t, timer);
    int value = gpio_get_value(dev->gpio);
    input_report_key(dev->input_dev, dev->key_code, !value);
    input_sync(dev->input_dev);
}

static irqreturn_t key_irq_handler(int irq, void *dev_id)
{
    struct key_dev *dev = (struct key_dev *)dev_id;
    mod_timer(&dev->timer, jiffies + msecs_to_jiffies(10));
    return IRQ_HANDLED;
}

static int key_probe(struct platform_device *pdev)
{
    struct device_node *np = pdev->dev.of_node;
    struct key_dev *dev;
    int ret;

    dev = devm_kzalloc(&pdev->dev, sizeof(*dev), GFP_KERNEL);
    if (!dev) return -ENOMEM;

    dev->gpio = of_get_named_gpio(np, "key-gpio", 0);
    if (!gpio_is_valid(dev->gpio)) return -EINVAL;

    ret = gpio_request(dev->gpio, "key");
    if (ret) return ret;
    gpio_direction_input(dev->gpio);

    dev->irq = gpio_to_irq(dev->gpio);
    ret = request_irq(dev->irq, key_irq_handler,
                      IRQF_TRIGGER_RISING | IRQF_TRIGGER_FALLING,
                      "key", dev);
    if (ret) goto err_gpio;

    dev->input_dev = devm_input_allocate_device(&pdev->dev);
    dev->input_dev->name = "key";
    dev->input_dev->evbit[0] = BIT_MASK(EV_KEY);
    set_bit(KEY_ENTER, dev->input_dev->keybit);

    ret = input_register_device(dev->input_dev);
    if (ret) goto err_irq;

    timer_setup(&dev->timer, key_timer_callback, 0);
    platform_set_drvdata(pdev, dev);
    return 0;

err_irq:
    free_irq(dev->irq, dev);
err_gpio:
    gpio_free(dev->gpio);
    return ret;
}

static int key_remove(struct platform_device *pdev)
{
    struct key_dev *dev = platform_get_drvdata(pdev);
    del_timer_sync(&dev->timer);
    free_irq(dev->irq, dev);
    gpio_free(dev->gpio);
    return 0;
}

static const struct of_device_id key_of_match[] = {
    { .compatible = "my,key" },
    {}
};
MODULE_DEVICE_TABLE(of, key_of_match);

static struct platform_driver key_driver = {
    .probe  = key_probe,
    .remove = key_remove,
    .driver = {
        .name = "key_driver",
        .of_match_table = key_of_match,
    },
};
module_platform_driver(key_driver);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("GPIO Key driver with interrupt and debounce");
```

**设备树节点**：

```dts
my_key {
    compatible = "my,key";
    key-gpio = <&gpx1 0 0>;
};
```

