# 中断处理与内核调试

## 1. Linux 中断子系统

### 1.1 中断概念与分类

中断是 CPU 暂停当前任务转而处理紧急事件的机制。在嵌入式驱动开发中，中断是外设与 CPU 通信的核心方式。

| 分类方式 | 类型 | 说明 |
|----------|------|------|
| 来源 | 硬件中断 | 外部硬件产生的电信号（如 GPIO、网卡收包） |
| 来源 | 软中断 | 软件指令触发 |
| 触发方式 | 边沿触发 | 上升沿/下降沿触发 |
| 触发方式 | 电平触发 | 高电平/低电平触发 |

### 1.2 中断号

中断号是一个整型数字，用于唯一标识一个中断源：

- **硬中断号（hwirq）**：硬件中断编号，由中断控制器（GIC）定义
- **Linux 中断号（virq）**：内核使用的虚拟中断号，通过 `irq_of_parse_and_map()` 从设备树获取

```c
struct device_node *np = dev->of_node;
int irq = irq_of_parse_and_map(np, 0);
```

### 1.3 request_irq / free_irq

```c
#include <linux/interrupt.h>

int request_irq(unsigned int irq, irq_handler_t handler,
                unsigned long flags, const char *name, void *dev_id);
void free_irq(unsigned int irq, void *dev_id);
```

| 参数 | 说明 |
|------|------|
| `irq` | 中断号 |
| `handler` | 中断处理函数，返回 `IRQ_HANDLED` |
| `flags` | `IRQF_TRIGGER_RISING`/`IRQF_TRIGGER_FALLING`/`IRQF_TRIGGER_HIGH`/`IRQF_TRIGGER_LOW`，可或 `IRQF_SHARED` |
| `name` | 显示在 `/proc/interrupts` |
| `dev_id` | 私有数据指针，共享中断时必须唯一 |

```c
irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    struct my_device *dev = (struct my_device *)dev_id;
    // 处理中断...
    return IRQ_HANDLED;
}

// probe 中注册
int ret = request_irq(dev->irq, my_irq_handler,
                       IRQF_TRIGGER_RISING, "my_device", dev);

// remove 中释放
free_irq(dev->irq, dev);
```

### 1.4 中断上下文限制

- **不能睡眠**：不能调用 `msleep()`、`down()`、`mutex_lock()` 等
- **不能访问用户空间**：`copy_to_user`/`copy_from_user` 不可用
- **不能调度**：不能主动调用 `schedule()`
- **应尽快完成**：长时间占用会阻塞其他中断

## 2. 中断底半部机制

### 2.1 tasklet

tasklet 运行在软中断上下文，不可睡眠。

```c
#include <linux/interrupt.h>

void my_tasklet_func(unsigned long data)
{
    struct my_device *dev = (struct my_device *)data;
    // 底半部处理...
}

DECLARE_TASKLET(my_tasklet, my_tasklet_func, (unsigned long)dev);

irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    tasklet_schedule(&my_tasklet);
    return IRQ_HANDLED;
}

// 清理
tasklet_kill(&my_tasklet);
```

### 2.2 workqueue

workqueue 运行在内核线程（进程上下文）中，**可以睡眠**。

```c
#include <linux/workqueue.h>

void my_work_func(struct work_struct *work)
{
    struct my_device *dev = container_of(work, struct my_device, my_work);
    // 可调用 msleep、mutex_lock 等
}

// 在设备结构体中
struct my_device {
    struct work_struct my_work;
};

INIT_WORK(&dev->my_work, my_work_func);

// 调度
schedule_work(&dev->my_work);

// 延迟执行
schedule_delayed_work(&dev->my_dwork, msecs_to_jiffies(100));

// 清理
cancel_work_sync(&dev->my_work);
cancel_delayed_work_sync(&dev->my_dwork);
```

### 2.3 threaded irq

```c
int ret = request_threaded_irq(irq,
    my_hard_irq_handler,   // 顶半部（可为 NULL）
    my_thread_func,        // 底半部——在线程中执行，可睡眠
    IRQF_TRIGGER_RISING | IRQF_ONESHOT,
    "my_device", dev);
```

### 2.4 选择原则

| 机制 | 上下文 | 可睡眠 | 适用场景 |
|------|--------|--------|----------|
| tasklet | 软中断 | 否 | 简单不可阻塞操作 |
| workqueue | 内核线程 | 是 | 需调用可能阻塞的 API |
| threaded irq | 内核线程 | 是 | 大部分工作在睡眠中完成 |

## 3. 内核调试技术

### 3.1 printk 打印级别

```c
#include <linux/kernel.h>

pr_emerg("system is dead\n");    // KERN_EMERG
pr_err("error: %d\n", err);      // KERN_ERR
pr_warn("warning\n");            // KERN_WARNING
pr_info("loaded\n");             // KERN_INFO
pr_debug("value=%d\n", val);     // KERN_DEBUG（需 #define DEBUG）
```

```bash
cat /proc/sys/kernel/printk   # 查看控制台级别
echo 8 > /proc/sys/kernel/printk  # 输出所有 printk
```

### 3.2 debugfs 使用

```c
#include <linux/debugfs.h>

struct dentry *debugfs_dir = debugfs_create_dir("my_driver", NULL);
debugfs_create_u32("counter", 0644, debugfs_dir, &dev->counter);

// remove 中
debugfs_remove_recursive(debugfs_dir);
```

```bash
mount -t debugfs none /sys/kernel/debug
cat /sys/kernel/debug/my_driver/counter
```

### 3.3 oops/panic 分析

```bash
# 反汇编定位
arm-linux-gnueabihf-objdump -d -S my_driver.ko > my_driver.dis

# 地址转源码行号
arm-linux-gnueabihf-addr2line -e vmlinux -f <函数地址>

# 查看内核日志
dmesg | tail -50
```

## 4. 项目示例：GPIO 按键中断驱动

```c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/interrupt.h>
#include <linux/input.h>
#include <linux/gpio.h>
#include <linux/of_gpio.h>
#include <linux/platform_device.h>
#include <linux/timer.h>

struct key_dev {
    int gpio;
    int irq;
    int key_code;
    struct input_dev *input_dev;
    struct timer_list timer;
};

static void key_timer_callback(struct timer_list *t)
{
    struct key_dev *dev = from_timer(dev, t, timer);
    int value = gpio_get_value(dev->gpio);
    input_report_key(dev->input_dev, dev->key_code, !value);
    input_sync(dev->input_dev);
}

static irqreturn_t key_irq_handler(int irq, void *dev_id)
{
    struct key_dev *dev = (struct key_dev *)dev_id;
    mod_timer(&dev->timer, jiffies + msecs_to_jiffies(10));
    return IRQ_HANDLED;
}

static int key_probe(struct platform_device *pdev)
{
    struct device_node *np = pdev->dev.of_node;
    struct key_dev *dev;
    int ret;

    dev = devm_kzalloc(&pdev->dev, sizeof(*dev), GFP_KERNEL);
    if (!dev) return -ENOMEM;

    dev->gpio = of_get_named_gpio(np, "key-gpio", 0);
    if (!gpio_is_valid(dev->gpio)) return -EINVAL;

    ret = gpio_request(dev->gpio, "key");
    if (ret) return ret;
    gpio_direction_input(dev->gpio);

    dev->irq = gpio_to_irq(dev->gpio);
    ret = request_irq(dev->irq, key_irq_handler,
                      IRQF_TRIGGER_RISING | IRQF_TRIGGER_FALLING,
                      "key", dev);
    if (ret) goto err_gpio;

    dev->input_dev = devm_input_allocate_device(&pdev->dev);
    dev->input_dev->name = "key";
    dev->input_dev->evbit[0] = BIT_MASK(EV_KEY);
    set_bit(KEY_ENTER, dev->input_dev->keybit);

    ret = input_register_device(dev->input_dev);
    if (ret) goto err_irq;

    timer_setup(&dev->timer, key_timer_callback, 0);
    platform_set_drvdata(pdev, dev);
    return 0;

err_irq:
    free_irq(dev->irq, dev);
err_gpio:
    gpio_free(dev->gpio);
    return ret;
}

static int key_remove(struct platform_device *pdev)
{
    struct key_dev *dev = platform_get_drvdata(pdev);
    del_timer_sync(&dev->timer);
    free_irq(dev->irq, dev);
    gpio_free(dev->gpio);
    return 0;
}

static const struct of_device_id key_of_match[] = {
    { .compatible = "my,key" }, {}
};
MODULE_DEVICE_TABLE(of, key_of_match);

static struct platform_driver key_driver = {
    .probe  = key_probe,
    .remove = key_remove,
    .driver = {
        .name = "key_driver",
        .of_match_table = key_of_match,
    },
};
module_platform_driver(key_driver);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("GPIO Key driver with interrupt and debounce");
```

**设备树**：

```dts
my_key {
    compatible = "my,key";
    key-gpio = <&gpx1 0 0>;
};
```
