## 1.主干代码

```asm
#include <asm-offsets.h>
#include <config.h>
#include <asm/system.h>
#include <linux/linkage.h>
#include <asm/armv7.h>
	.globl	reset
	.globl	save_boot_params_ret
	.type   save_boot_params_ret,%function @将save_boot_params_ret定义为一个函数
#ifdef CONFIG_ARMV7_LPAE
	.global	switch_to_hypervisor_ret
#endif

@=============================================================================================================
/*
*CONFIG_IS_ENABLED(TINY_FRAMEWORK) 不是简单的 #ifdef CONFIG_TINY_FRAMEWORK。
*在 U-Boot 中，这个宏会根据当前正在编译哪个阶段（U-Boot 本体、SPL 还是 TPL），自动查找对应的配置项。
*对于 TPL 阶段，它实际检查的是 CONFIG_TPL_TINY_FRAMEWORK。
*因此 #if !CONFIG_IS_ENABLED(...) 在编译 TPL 时，等价于判断 CONFIG_TPL_TINY_FRAMEWORK 是否未被设置。
*/
#if !CONFIG_IS_ENABLED(TINY_FRAMEWORK)
reset:
	/* Allow the board to save important registers */ @就是BootROM加载后续启动阶段时的堆栈寄存器
	b	save_boot_params
save_boot_params_ret:
@=============================================================================================================
#ifdef CONFIG_ARMV7_LPAE
/*
 * check for Hypervisor support
 */
	mrc	p15, 0, r0, c0, c1, 1		@ read ID_PFR1
	and	r0, r0, #CPUID_ARM_VIRT_MASK	@ mask virtualization bits
	cmp	r0, #(1 << CPUID_ARM_VIRT_SHIFT)
	beq	switch_to_hypervisor
switch_to_hypervisor_ret:
#endif
@=============================================================================================================
/*
* disable interrupts (FIQ and IRQ), also set the cpu to SVC32 mode,
* except if in HYP mode already
*/
	mrs	r0, cpsr
	and	r1, r0, #0x1f		@ mask mode bits
	teq	r1, #0x1a			@ test for HYP mode
	bicne	r0, r0, #0x1f	 @ clear all mode bits
	orrne	r0, r0, #0x13	 @ set SVC mode
	orr	r0, r0, #0xc0		 @ disable FIQ and IRQ
	msr	cpsr,r0
@=============================================================================================================
	/* Enable ACTLR.SMP bit 开启SMP缓存一致性（coherency）
	*CPU1 修改共享数据后，CPU0 不能一直使用过期的缓存内容。
	*ACTLR.SMP 在相应的处理器实现上，表示当前这个 CPU 核按多核一致性的规则参与工作。
	*因此，使用多个核时，通常每个参与的核都要完成相应设置。
	*/
	mrc	p15, 0, r0, c1, c0, 1
	orr	r0, r0, #(1 << 6)	@ Enable ACTLR.SMP bit
	mcr	p15, 0, r0, c1, c0, 1
@=============================================================================================================
/*
 * Setup vector:
 * (OMAP4 spl TEXT_BASE is not 32 byte aligned.
 * Continue to use ROM code vector only in OMAP4 spl)
 */
#if !(defined(CONFIG_OMAP44XX) && defined(CONFIG_SPL_BUILD))
	/* Set V=0 in CP15 SCTLR register - for VBAR to point to vector */
	mrc	p15, 0, r0, c1, c0, 0	@ Read CP15 SCTLR Register,读取 SCTLR
	bic	r0, #CR_V @ V = 0,SCTLR.V 控制是否使用固定的高地址异常向量。为 0 即通过 VBAR 指定异常向量表的起始地址。
	mcr	p15, 0, r0, c1, c0, 0	@ Write CP15 SCTLR Register,写回 SCTLR

	/* Set vector address in CP15 VBAR register */
	ldr	r0, =_start
	mcr	p15, 0, r0, c12, c0, 0	@Set VBAR
#endif
@=============================================================================================================
	/* Enable Asynchronous external abort after vectors setup */
	/*
	*CPSR.A 是异步 Abort 的屏蔽位：1 表示屏蔽，0 表示不屏蔽。
	*在设置向量表之后是因为一旦允许响应这类异常，CPU 就可能跳到 Abort 异常入口；
	*必须先让那个入口指向可用的处理代码。
	*异步表示异常可能在造成问题的总线访问之后才报告，因此异常发生时的指令不一定就是出错的那条指令。
	*/
	mrs	r0, cpsr
	bic	r0, r0, #0x100		@ CPSR.A bit
	msr	cpsr_x,r0
@=============================================================================================================
	/* the mask ROM code should have PLL and others stable */
	/*如果芯片先执行了内部 Mask ROM，ROM 代码可能已经把 PLL 等硬件设置稳定。
	*因此某些板子会通过这些配置宏跳过重复的底层初始化.
	*/
#ifndef CONFIG_SKIP_LOWLEVEL_INIT
	bl	cpu_init_cp15 				@调用 CP15 相关初始化
#ifndef CONFIG_SKIP_LOWLEVEL_INIT_ONLY
	bl	cpu_init_crit				@调用关键的底层初始化
#endif
#endif  @--#if !CONFIG_IS_ENABLED(TINY_FRAMEWORK)
@=============================================================================================================
bl	_main
```

### 2.cpu_init_cp15

```asm
/*************************************************************************
 *
 * cpu_init_cp15
 *
 * Setup CP15 registers (cache, MMU, TLBs). The I-cache is turned on unless
 * CONFIG_SYS_ICACHE_OFF is defined.
 *
 *************************************************************************/
ENTRY(cpu_init_cp15)
	/*
	 * Invalidate L1 I/D
	 */
	mov	r0, #0			@ set up for MCR
	mcr	p15, 0, r0, c8, c7, 0	@ invalidate TLBs
	mcr	p15, 0, r0, c7, c5, 0	@ invalidate icache
	mcr	p15, 0, r0, c7, c5, 6	@ invalidate BP array
	mcr     p15, 0, r0, c7, c10, 4	@ DSB
	mcr     p15, 0, r0, c7, c5, 4	@ ISB

	/*
	 * disable MMU stuff and caches
	 */
	mrc	p15, 0, r0, c1, c0, 0
	bic	r0, r0, #0x00002000	@ clear bits 13 (--V-)
	bic	r0, r0, #0x00000007	@ clear bits 2:0 (-CAM)
#if 0 /* There is unalign access when decompress firmware. */
	orr	r0, r0, #0x00000002	@ set bit 1 (--A-) Align
#endif
	orr	r0, r0, #0x00000800	@ set bit 11 (Z---) BTB
#ifdef CONFIG_SYS_ICACHE_OFF
	bic	r0, r0, #0x00001000	@ clear bit 12 (I) I-cache
#else
	orr	r0, r0, #0x00001000	@ set bit 12 (I) I-cache
#endif
	mcr	p15, 0, r0, c1, c0, 0
@=============================================================================================================
	/*
	*提取 CPU 核的版本号,供下面的规避编号430973代码使用
	*/
	
	mov	r5, lr			@ Store my Caller
	mrc	p15, 0, r1, c0, c0, 0	@ r1 has Read Main ID Register (MIDR)
	mov	r3, r1, lsr #20		@ get variant field
	and	r3, r3, #0xf		@ r3 has CPU variant
	and	r4, r1, #0xf		@ r4 has CPU revision
	mov	r2, r3, lsl #4		@ shift variant field for combined value
	orr	r2, r4, r2		@ r2 has combined CPU variant + revision
	
	/*
	*ARM CPU 勘误（errata）的规避代码。某些 CPU 核的特定版本存在硬件问题，启用对应的 CONFIG_ARM_ERRATA_... 后，编译器才会把	 *规避指令放进 U-Boot。
	*这组操作用于规避编号 716044 的问题
	*/
#ifdef CONFIG_ARM_ERRATA_430973
	cmp	r2, #0x21		@ Only on < r2p1
	bge	skip_errata_430973

	mrc	p15, 0, r0, c1, c0, 1	@ Read ACR
	orr	r0, r0, #(0x1 << 6)	@ Set IBE bit
	push	{r1-r5}			@ Save the cpu info registers
	bl	v7_arch_cp15_set_acr
	pop	{r1-r5}			@ Restore the cpu info - fall through
skip_errata_430973:
#endif
@=============================================================================================================
#if defined(CONFIG_ARM_ZERO_CNTVOFF) && defined(CONFIG_SPL_BUILD)
	/*
	 *只有在以下条件之一满足时，才能访问 CNTVOFF：
     *1. CPU 处于 Hyp 模式；
     *2. CPU 处于 Monitor 模式，且 SCR.NS = 1。
     *清零操作放在 SPL 的早期启动代码里：此时还有机会处理 CPU 的特权模式和安全状态
	 */
	/* switch to MON */
	cps #22
	isb

	/* Update SCR.NS to non Secure mode */
	mrc	p15, 0, r0, c1, c1, 0
	orr r0, r0, #(1 << 0)
	mcr	p15, 0, r0, c1, c1, 0
	isb

	/* set vtimer virtual offset 0 */
	mov	r0, #0
	mcrr	p15, 4, r0, r0, c14	@ CNTVOFF

	/* Update SCR.NS to Secure mode */
	mrc	p15, 0, r0, c1, c1, 0
	bic r0, r0, #(1 << 0)
	mcr	p15, 0, r0, c1, c1, 0
	isb

	/* switch back to SVC */
	cps #19
	isb
#endif
	mov	pc, r5			@ back to my caller
ENDPROC(cpu_init_cp15)
```
### 3.lowlevel_init

```asm
#include <asm-offsets.h>
#include <config.h>
#include <linux/linkage.h>

.pushsection .text.s_init, "ax"
WEAK(s_init)
	bx	lr
ENDPROC(s_init)
.popsection

@给 lowlevel_init 单独起了一个段名，方便链接器以后单独找到、安排或删除这个函数。.pushsection 是开始把代码放进这个段；.popsection 是这个函数写完后，恢复原来的段，避免后续代码也被归进去。
.pushsection .text.lowlevel_init, "ax"
WEAK(lowlevel_init)
	/*
	 * Setup a temporary stack. Global data is not available yet.
	 */
#if defined(CONFIG_SPL_BUILD) && defined(CONFIG_SPL_STACK)
	ldr	sp, =CONFIG_SPL_STACK
#else
	ldr	sp, =CONFIG_SYS_INIT_SP_ADDR
#endif
	bic	sp, sp, #7 /* 8-byte alignment for ABI compliance */
#ifdef CONFIG_SPL_DM
	mov	r9, #0
#else
	/*
	 * Set up global data for boards that still need it. This will be
	 * removed soon.
	 */
#ifdef CONFIG_SPL_BUILD
	ldr	r9, =gdata
#else
	sub	sp, sp, #GD_SIZE
	bic	sp, sp, #7
	mov	r9, sp
#endif
#endif
	/*
	 * Save the old lr(passed in ip) and the current lr to stack
	 */
	push	{ip, lr}

	/*
	 * Call the very early init function. This should do only the
	 * absolute bare minimum to get started. It should not:
	 *
	 * - set up DRAM
	 * - use global_data
	 * - clear BSS
	 * - try to start a console
	 *
	 * For boards with SPL this should be empty since SPL can do all of
	 * this init in the SPL board_init_f() function which is called
	 * immediately after this.
	 */
	bl	s_init
	pop	{ip, pc}
ENDPROC(lowlevel_init)
.popsection
```

## 3.从 `s_init` 到 `_main`

`s_init` 比 `_main` 更早执行，只允许做绝对必要的初始化：此时不能依赖 DRAM、GD（`global_data`）、BSS 或控制台。对于采用 SPL 的板子，通用注释建议 `s_init` 为空，把早期初始化放在 SPL 的 `board_init_f()` 中。调用前的 `push {ip, lr}` 保存 `ip` 中传来的旧返回地址和当前返回地址；`bl s_init` 会改写 `lr`；`pop {ip, pc}` 恢复 `ip` 并返回。

`start.S` 跳入 `crt0` 的 `_main` 后，按以下顺序建立 C 运行环境：

1. 在当时可用的内存中准备临时栈和 GD，并在调用 `board_init_f()` 前清零 GD。GD 是专门准备的早期数据结构；普通全局**变量**和 BSS 此时仍不可用。
2. 调用 `board_init_f()` 做早期硬件准备，并在 GD 中记录后续需要的地址，例如新栈、GD 位置和重定位目标。具体是否需要初始化 DDR，由当前启动阶段决定。
3. As system RAM may not be available yet, , board_init_f() must use the current GD tostore any data which must be passed on to later stages.注释中“DRAM 可能尚不可用”是通用约束：有的板子由 SPL 在 SRAM 中初始化 DDR。若当前 RK3506 启动镜像已将 SPL 加载到 `0x03f00000` 执行，DDR 必须在此之前可用；具体由哪个前级程序初始化，应以实际镜像为准。临时栈和 GD 的位置仍须看具体配置，不能由 SPL 代码地址推断。
4. U-Boot 本体切换到安排好的栈和 GD，再调用 `relocate_code()` 搬移自身；SPL 不进行代码重定位。
5. 清零 BSS，建立可写数据等完整 C 环境，然后进入 `board_init_r()`；U-Boot 本体还会在此之前执行需要的 CPU 运行环境设置。SPL 是否搬移栈和 GD，取决于 `CONFIG_SPL_STACK_R` 等配置。

# ctr0.S

## 1.建立初始环境

为调用 `board_init_f()` 建立初始环境。这个环境只提供一个栈和一个用于存放 GD（“global data”，全局数据）结构的位置，两者都位于某种可立即使用的 RAM 中（SRAM、锁定的 cache 等）。在此上下文中，可变的全局数据（无论是已初始化的还是未初始化的（BSS））都**不可用**；只有**常量型的已初始化数据**可用。在调用 `board_init_f()` 之前，GD 应被清零。

```asm
ENTRY(_main)

/*
 * Set up initial C runtime environment and call board_init_f(0).
 */

#if defined(CONFIG_TPL_BUILD) && defined(CONFIG_TPL_NEEDS_SEPARATE_STACK)
	ldr	r0, =(CONFIG_TPL_STACK)
#elif defined(CONFIG_SPL_BUILD) && defined(CONFIG_SPL_STACK)
	ldr	r0, =(CONFIG_SPL_STACK)
#else
	ldr	r0, =(CONFIG_SYS_INIT_SP_ADDR)
#endif
	bic	r0, r0, #7	/* 8-byte alignment for ABI compliance */
	mov	sp, r0
	bl	board_init_f_alloc_reserve
	mov	sp, r0
	/* set up gd here, outside any C code */
	mov	r9, r0
	bl	board_init_f_init_reserve
	bl	board_init_f_boot_flags

	bl	board_init_f
```

## 2.U-Boot重定位

### 1.拷贝搬移

```shell
arch\arm\lib\relocate.S
```

主要任务就是把当前运行位置的代码复制到目标地址，并修正代码中保存的地址。它处理的是内存地址，不是在搬文件。

- `r0` 是目标起始地址；`__image_copy_start` 是当前起始地址。`r4 = r0 - __image_copy_start` 算出两处相差多少。如果差值为 0，就不用搬。
- `copy_loop` 每次复制两个寄存器，即 8 字节，直到 `__image_copy_end`。
- `fixloop` 遍历 `.rel.dyn` 重定位表。遇到 `R_ARM_RELATIVE`，先把“需要修改的位置”加上 `r4`，再把那个位置存放的地址值也加上 `r4`。这样复制后的代码仍能指向复制后的数据。

  ```
  例如代码从 `0x1000` 搬到 0x8000，差值是 0x7000。原来位于 0x1100 的一个指针，内容为 0x1200；重定位后，会在 0x8100 处把它改成 0x8200。
  ```
- 最后在 XScale 平台做缓存处理，并通过 `bx lr` 返回调用者。**这段函数本身没有跳转到新地址执行**；后续由调用者安排。

- CPU 的程序计数器 `PC` 和返回地址 `LR` 不会因为复制而自动加上偏移量。`relocate_code` 最后的 `bx lr` 通常仍返回** **LR 当时保存的地址****。
- 调用者随后需要显式跳转到新地址，CPU 才开始执行新副本。

  ```
  例如调用发生在旧地址 0x1100，整体搬移偏移是 0x700。新副本位于 0x8100，但 bx lr 仍按原来的返回地址回到 0x1100 附近；只有后续跳转，执行位置才会变成 0x8100 附近。具体跳转发生在哪一行，要看调用 relocate_code 的代码。
  ```

### 2.跳转执行

```asm
#if ! defined(CONFIG_SPL_BUILD)

/*
 * Set up intermediate environment (new sp and gd) and call
 * relocate_code(addr_moni). Trick here is that we'll return
 * 'here' but relocated.
 */

	ldr	r0, [r9, #GD_START_ADDR_SP]	/* sp = gd->start_addr_sp */
	bic	r0, r0, #7	/* 8-byte alignment for ABI compliance */
	mov	sp, r0
	ldr	r9, [r9, #GD_BD]		/* r9 = gd->bd */
	sub	r9, r9, #GD_SIZE		/* new GD is below bd */

#ifndef CONFIG_SKIP_RELOCATE_UBOOT
	adr	lr, here					@取here旧位置
	ldr	r0, [r9, #GD_RELOC_OFF]		/* r0 = gd->reloc_off */
	add	lr, lr, r0                    @lr存的是here新位置
#if defined(CONFIG_CPU_V7M)
	orr	lr, #1				/* As required by Thumb-only */
#endif
	ldr	r0, [r9, #GD_RELOCADDR]		/* r0 = gd->relocaddr */
	b	relocate_code
#endif
here:
```

`relocate_code` 复制代码后执行 `bx lr`，于是跳到**新副本里的 `here`**。例如旧 `here` 在 `0x1200`，偏移是 `0x7000`，那么 LR 被设为 `0x8200`；`bx lr` 后 CPU 就从 `0x8200` 执行。旧代码并没有消失，只是 CPU 切换到了新副本。随后从新位置执行 `relocate_vectors`、初始化运行环境、清零 BSS，最后跳到 `board_init_r`。这里用 `b relocate_code` 而不是 `bl` 很关键：`bl` 会重新写入 LR，破坏预先算好的新地址。
