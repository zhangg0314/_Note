# 单片机链接脚本

```asm
OUTPUT_FORMAT("elf32-littlearm", "elf32-littlearm", "elf32-littlearm")
/*指定输出文件的格式为 32 位 ARM 小端模式 ELF 格式
三个参数分别对应默认格式、大端格式、小端格式（这里都设为小端）*/
OUTPUT_ARCH(arm)#指定输出文件的目标架构为 ARM
ENTRY(_start)#指定程序的入口点为_start符号，这是程序开始执行的位置
SECTIONS#开始定义段（sections）的布局，这是链接脚本的核心部分
{
	. = 0x00000000;#设置当前地址计数器（.）为 0x00000000，即程序加载的起始地址

	. = ALIGN(4);#对齐到 4 字节边界
	.text :
	{
		__image_copy_start = .;#记录__image_copy_start 为当前地址（镜像复制的起始点）
		CPUDIR/start.o (.text*)#首先放入 CPU 相关目录下的 start.o 文件的所有.text 段
		*(.text*)#然后放入所有其他目标文件的.text 段
	}

	. = ALIGN(4);
	.rodata : { *(SORT_BY_ALIGNMENT(SORT_BY_NAME(.rodata*))) }#按名称和对齐方式排序所有.rodata 段并放入

	. = ALIGN(4);
	.data : {
		*(.data*)#包含所有目标文件的.data 段
	}

	. = ALIGN(4);

	. = .;

	. = ALIGN(4);#多次对齐到 4 字节边界（可能是为了兼容性或后续修改预留）
	.u_boot_list : {
	#include <u-boot.lst>#包含 u-boot.lst 文件中定义的各种 U-Boot 链表结构
	}

	. = ALIGN(4);

	__image_copy_end = .;#记录__image_copy_end 为当前地址（镜像复制的结束点）

	.rel.dyn : {
		__rel_dyn_start = .;
		*(.rel*)#包含所有重定位相关的段
		__rel_dyn_end = .;#记录重定位信息的起始和结束地址
	}

	.dynsym : {
		__dynsym_start = .;#记录动态符号表的起始地址
		*(.dynsym)#包含所有动态符号信息
	}

	_end = .;#定义_end 符号为当前地址，标记程序的结束位置

	/*
	 * Deprecated: this MMU section is used by pxa at present but
	 * should not be used by new boards/CPUs.
	 */
	. = ALIGN(4096);#按 4096 字节（页大小）对齐
	.mmutable : {
		*(.mmutable)#注释说明这是过时的，仅用于 pxa 处理器，新板不应该使用
	}

	.bss __rel_dyn_start (OVERLAY) : {
		__bss_start = .;
		*(.bss*)
		 . = ALIGN(4);
		__bss_end__ = .;
	}

	/DISCARD/ : { *(.dynstr*) }
	/DISCARD/ : { *(.dynamic*) }
	/DISCARD/ : { *(.plt*) }
	/DISCARD/ : { *(.interp*) }
	/DISCARD/ : { *(.gnu*) }
}

#if defined(CONFIG_SPL_TEXT_BASE) && defined(CONFIG_SPL_MAX_SIZE)
ASSERT(__bss_end__ < (CONFIG_SPL_TEXT_BASE + CONFIG_SPL_MAX_SIZE), "SPL image too big");
#使用 ASSERT 检查 SPL 镜像大小是否超过限制,如果__bss_end__超过了最大允许地址，则编译时会报错 "SPL image too big"
#endif
```
